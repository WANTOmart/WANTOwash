// Helper terpisah untuk merapikan logika ekstraksi nama paket function formatNamaPaket(paket) { if (Array.isArray(paket)) { return paket .map(p => (typeof p === 'object' && p !== null) ? (p.nama || p.namaPaket || '-') : p) .filter(Boolean) .join(", ") || "-"; } if (typeof paket === 'object' && paket !== null) { return paket.nama || paket.namaPaket || "-"; } if (typeof paket === 'string' && paket.trim() !== "") { return paket; } return "-"; } function tampilLaporan(listData) { // Memastikan listData selalu berbentuk Array aman const data = Array.isArray(listData) ? listData : []; let totalTabel = 0; let html = ""; if (data.length === 0) { html = `Tidak ada data transaksi yang sesuai.`; } else { html = data.map((item, index) => { const totalNominal = Number(item.total) || 0; totalTabel += totalNominal; const namaPaket = formatNamaPaket(item.paket); return ` ${index + 1} ${item.nomor || item.noTrx || '-'} ${item.tanggal || '-'} ${item.pelanggan || item.namaPelanggan || '-'} ${item.motor || item.namaMotor || '-'} ${namaPaket} Rp ${totalNominal.toLocaleString("id-ID")} `; }).join(''); } // Format mata uang sekali untuk efisiensi const formattedTotal = "Rp " + totalTabel.toLocaleString("id-ID"); // Rendering ke DOM document.getElementById("tabelLaporan").innerHTML = html; document.getElementById("grandTotalTabel").innerHTML = formattedTotal; document.getElementById("sumFilter").innerHTML = formattedTotal; document.getElementById("countFilter").innerHTML = ` ${data.length} Transaksi`; }
