function tampilLaporan(listData) { let html = ""; let totalTabel = 0; if (!listData || listData.length === 0) { html = `Tidak ada data transaksi yang sesuai.`; } else { listData.forEach(function(item, index) { let totalNominal = Number(item.total) || 0; totalTabel += totalNominal; // --- PROTEKSI & FORMATTING PAKET CUCI --- let namaPaket = "-"; if (Array.isArray(item.paket)) { // Jika paket berupa array dari string/objek (misal: beberapa paket sekaligus) namaPaket = item.paket.map(p => (typeof p === 'object' && p !== null) ? (p.nama || p.namaPaket || '-') : p).join(", "); } else if (typeof item.paket === 'object' && item.paket !== null) { // Jika paket disimpan sebagai Objek namaPaket = item.paket.nama || item.paket.namaPaket || "-"; } else if (typeof item.paket === 'string' && item.paket.trim() !== "") { // Jika paket berupa String biasa namaPaket = item.paket; } html += ` ${index + 1} ${item.nomor || item.noTrx || '-'} ${item.tanggal || '-'} ${item.pelanggan || item.namaPelanggan || '-'} ${item.motor || item.namaMotor || '-'} ${namaPaket} Rp ${totalNominal.toLocaleString("id-ID")} `; }); } document.getElementById("tabelLaporan").innerHTML = html; document.getElementById("grandTotalTabel").innerHTML = "Rp " + totalTabel.toLocaleString("id-ID"); // Set Card Ringkasan Filter document.getElementById("sumFilter").innerHTML = "Rp " + totalTabel.toLocaleString("id-ID"); document.getElementById("countFilter").innerHTML = ` ${listData.length} Transaksi`; }
